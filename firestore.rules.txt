rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    function isAdmin() {
      return isAuthenticated() && (
        request.auth.token.email == 'fundishaadmin@gmail.com' ||
        request.auth.token.admin == true
      );
    }

    // Public Educational Resources
    match /resources/{resourceId} {
      // Anyone can read resources
      allow read: if true;
      // Admins can manage resources; allow counter increments for views/downloads
      allow write: if isAdmin() || (
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['viewCount', 'downloadCount', 'popularity', 'updatedAt'])
      );
    }

    // UNEB Quizzes
    match /quizzes/{quizId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Student Profiles & Gamification
    match /users/{userId} {
      // Read allowed for profile access and leaderboard public views
      allow read: if true;
      // User can create, update, or delete only their own record
      allow create: if isOwner(userId);
      allow update: if isOwner(userId) || isAdmin();
      allow delete: if isOwner(userId) || isAdmin();

      // Nested subcollections strictly owned by the student
      match /bookmarks/{bookmarkId} {
        allow read, write: if isOwner(userId);
      }

      match /quizAttempts/{attemptId} {
        allow read, write: if isOwner(userId);
      }

      match /recentlyViewed/{viewId} {
        allow read, write: if isOwner(userId);
      }

      match /progress/{progressId} {
        allow read, write: if isOwner(userId);
      }
    }

    // Content Quality and Broken Link Reports
    match /reports/{reportId} {
      // Students can submit reports
      allow create: if true;
      // Only platform admins can view, review, and manage reports
      allow read, update, delete: if isAdmin();
    }
  }
}
